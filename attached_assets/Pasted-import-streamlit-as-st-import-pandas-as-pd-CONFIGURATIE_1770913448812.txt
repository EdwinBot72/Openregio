import streamlit as st
import pandas as pd

# --- CONFIGURATIE ---
st.set_page_config(page_title="OpenRegio Dashboard", layout="wide")

# --- CUSTOM CSS VOOR MENSELIJKE LOOK ---
st.markdown("""
    <style>
    .main { background-color: #f9f9f9; }
    .stButton>button { border-radius: 20px; border: 1px solid #ff4b4b; background-color: white; color: #ff4b4b; }
    .stButton>button:hover { background-color: #ff4b4b; color: white; }
    </style>
    """, unsafe_content_type=True)

# --- SIDEBAR (DE BUURMAN) ---
with st.sidebar:
    st.title("👋 Hallo Ondernemer!")
    st.write("Ik ben je digitale buurman. Waar kan ik je vandaag mee helpen?")
    keuze = st.radio("Wat wil je doen?", ["Kansen in de regio", "Lokaal samenwerken", "Mijn Impact"])

# --- HOOFDPAGINA ---
if keuze == "Kansen in de regio":
    st.header("🔍 RegioBot: Wat gebeurt er bij de gemeente?")
    st.info("Ik heb de laatste WOO-documenten gescand. Hier is wat relevant is voor jou:")
    
    # Voorbeeld data van de bot
    kansen = [
        {"Datum": "12-02", "Onderwerp": "Herinrichting Marktplein", "Kans voor": "Horeca & Groenvoorziening"},
        {"Datum": "10-02", "Onderwerp": "Subsidie Verduurzaming Bedrijfspanden", "Kans voor": "Alle ondernemers"},
    ]
    st.table(pd.DataFrame(kansen))
    
    vraag = st.text_input("Stel een vraag aan de RegioBot (bijv: 'Is er nieuws over parkeertarieven?')")
    if vraag:
        st.write(f"Vriendelijke RegioBot: 'Ik ga voor je kijken wat er in de stukken staat over *{vraag}*...'")

elif keuze == "Lokaal samenwerken":
    st.header("🤝 De RegioMarkt")
    st.subheader("Wie heb je nodig?")
    
    col1, col2 = st.columns(2)
    with col1:
        sector = st.selectbox("Sector", ["Bouw", "IT & Marketing", "Horeca", "Advies"])
    with col2:
        st.write(" ")
        st.button("Vind lokale partners")
    
    st.write("---")
    st.write("💡 **Tip:** Bakker Jan heeft nog een klusje voor een schilder. Iets voor jou?")

elif keuze == "Mijn Impact":
    st.header("📈 Jouw Regionale Voetafdruk")
    st.write("Hoeveel van jouw euro's blijven in de buurt?")
    
    omzet_lokaal = st.slider("Hoeveel procent koop jij lokaal in?", 0, 100, 25)
    impact = omzet_lokaal * 1.5 # Simpele factor voor lokale circulatie
    
    st.metric(label="Regionale Groeifactor", value=f"{impact}%", delta="Boven gemiddelde")
    st.write(f"Door {omzet_lokaal}% lokaal te besteden, help je direct 3 banen in de regio te behouden.")

# --- FOOTER ---
st.write("---")
st.caption("OpenRegio - Samen bouwen aan een menselijke economie.")
