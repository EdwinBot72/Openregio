Helemaal goed. Hieronder staat een **super minimal Replit-ready** versie:

✅ FastAPI
✅ haalt 1 pagina TenderNed op
✅ filtert op gemeente-naam (bijv. “Haarlemmermeer”)
✅ geen database (dus simpel)
✅ endpoint voor jouw dashboard cards: `/top?gemeente=haarlemmermeer`

### 3) `main.py`

```python
from fastapi import FastAPI, Query
import requests
from datetime import datetime, date

app = FastAPI(title="OpenRegio Minimal TenderNed API")

TNS_URL = "https://www.tenderned.nl/papi/tenderned-rs-tns/v2/publicaties"
UA = {"User-Agent": "OpenRegio-Minimal/0.1"}

def parse_iso_date(s: str | None) -> str | None:
    if not s:
        return None
    try:
        # handles "Z" timezone
        dt = datetime.fromisoformat(s.replace("Z", "+00:00"))
        return dt.date().isoformat()
    except Exception:
        return None

def days_left(yyyy_mm_dd: str | None) -> int | None:
    if not yyyy_mm_dd:
        return None
    try:
        d = date.fromisoformat(yyyy_mm_dd)
        return (d - date.today()).days
    except Exception:
        return None

def get_items(page: int = 0, size: int = 50) -> list[dict]:
    r = requests.get(TNS_URL, params={"page": page, "size": size}, headers=UA, timeout=20)
    r.raise_for_status()
    data = r.json()
    # TenderNed response key can differ; these are common
    return data.get("publicaties") or data.get("items") or data.get("content") or []

def pick(pub: dict, keys: list[str], default=None):
    for k in keys:
        v = pub.get(k)
        if v is not None and v != "":
            return v
    return default

def buyer_name(pub: dict) -> str:
    # common nested structures
    ad = pub.get("aanbestedendeDienst") or pub.get("koper") or {}
    if isinstance(ad, dict):
        return ad.get("naam") or ""
    if isinstance(ad, str):
        return ad
    return ""

@app.get("/top")
def top_cards(
    gemeente: str = Query(..., description="Bijv: haarlemmermeer"),
    limit: int = Query(5, ge=1, le=10),
    page: int = Query(0, ge=0),
    size: int = Query(50, ge=1, le=200),
):
    gemeente_lc = gemeente.strip().lower()

    pubs = get_items(page=page, size=size)

    cards = []
    for p in pubs:
        bname = buyer_name(p)
        if gemeente_lc not in (bname or "").lower():
            continue

        title = pick(p, ["titel", "title"], "Onbekende titel")
        deadline_iso = pick(p, ["sluitingsdatum", "deadline"], None)
        deadline = parse_iso_date(deadline_iso)

        # budget is often missing/varies; keep it optional
        budget = p.get("raming") or p.get("geraamdeWaarde") or None
        budget_min = None
        budget_max = None
        if isinstance(budget, (int, float)):
            budget_min = int(budget)
            budget_max = int(budget)
        elif isinstance(budget, dict):
            budget_min = budget.get("min") or budget.get("minimum")
            budget_max = budget.get("max") or budget.get("maximum")
            if isinstance(budget_min, float): budget_min = int(budget_min)
            if isinstance(budget_max, float): budget_max = int(budget_max)

        cards.append({
            "title": title,
            "buyer": bname,
            "budget_min": budget_min,
            "budget_max": budget_max,
            "deadline": deadline,
            "days_left": days_left(deadline),
            "match_score": 0.8,  # MVP fixed score, later echt berekenen
            "source": "TenderNed",
            "source_url": pick(p, ["url", "publicatieUrl"], None),
        })

        if len(cards) >= limit:
            break

    return {
        "gemeente": gemeente,
        "count": len(cards),
        "items": cards
    }

@app.get("/")
def home():
    return {
        "ok": True,
        "usage": {
            "top": "/top?gemeente=haarlemmermeer"
        }
    }
```
