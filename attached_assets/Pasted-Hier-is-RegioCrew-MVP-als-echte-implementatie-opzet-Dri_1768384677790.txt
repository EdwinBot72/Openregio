1) DB: Drizzle schema (shared/schema.ts)

Plak dit erbij (namen zijn consistent met jouw stijl). Gebruik gen_random_uuid().

// shared/schema.ts
import { pgTable, uuid, text, boolean, timestamp, jsonb, numeric, integer } from "drizzle-orm/pg-core";
import { users, regions, businesses } from "./schema"; // pas aan aan jouw exports

export const crewProfiles = pgTable("crew_profiles", {
  id: uuid("id").defaultRandom().primaryKey(),
  userId: uuid("user_id").notNull().references(() => users.id, { onDelete: "cascade" }),
  regionId: uuid("region_id").notNull().references(() => regions.id, { onDelete: "cascade" }),

  displayName: text("display_name").notNull(),
  headline: text("headline"),
  categories: jsonb("categories").$type<string[]>().notNull().default([]),
  skills: jsonb("skills").$type<string[]>().notNull().default([]),

  availability: jsonb("availability").$type<Record<string, any>>().notNull().default({}),
  rateType: text("rate_type").notNull().default("hour"), // hour|day|fixed
  rateMinEur: numeric("rate_min_eur", { precision: 12, scale: 2 }),

  phone: text("phone"),
  bio: text("bio"),

  isActive: boolean("is_active").notNull().default(true),

  createdAt: timestamp("created_at", { withTimezone: true }).notNull().defaultNow(),
  updatedAt: timestamp("updated_at", { withTimezone: true }).notNull().defaultNow(),
});

export const crewRequests = pgTable("crew_requests", {
  id: uuid("id").defaultRandom().primaryKey(),
  businessId: uuid("business_id").notNull().references(() => businesses.id, { onDelete: "cascade" }),
  regionId: uuid("region_id").notNull().references(() => regions.id, { onDelete: "cascade" }),

  title: text("title").notNull(),
  description: text("description"),
  category: text("category").notNull(), // retail|horeca|etc
  skills: jsonb("skills").$type<string[]>().notNull().default([]),

  startAt: timestamp("start_at", { withTimezone: true }).notNull(),
  endAt: timestamp("end_at", { withTimezone: true }).notNull(),

  rateType: text("rate_type").notNull().default("negotiable"), // hour|day|fixed|negotiable
  rateEur: numeric("rate_eur", { precision: 12, scale: 2 }),

  locationText: text("location_text"),

  status: text("status").notNull().default("open"), // open|closed|filled|cancelled

  createdAt: timestamp("created_at", { withTimezone: true }).notNull().defaultNow(),
  updatedAt: timestamp("updated_at", { withTimezone: true }).notNull().defaultNow(),
});

export const crewApplications = pgTable("crew_applications", {
  id: uuid("id").defaultRandom().primaryKey(),
  requestId: uuid("request_id").notNull().references(() => crewRequests.id, { onDelete: "cascade" }),
  crewProfileId: uuid("crew_profile_id").notNull().references(() => crewProfiles.id, { onDelete: "cascade" }),

  message: text("message"),
  status: text("status").notNull().default("applied"), // applied|shortlisted|accepted|rejected|withdrawn

  createdAt: timestamp("created_at", { withTimezone: true }).notNull().defaultNow(),
  updatedAt: timestamp("updated_at", { withTimezone: true }).notNull().defaultNow(),
});

// Reviews (optioneel voor MVP, maar slim om meteen te hebben)
export const crewReviews = pgTable("crew_reviews", {
  id: uuid("id").defaultRandom().primaryKey(),
  requestId: uuid("request_id").notNull().references(() => crewRequests.id, { onDelete: "cascade" }),
  fromBusinessId: uuid("from_business_id").notNull().references(() => businesses.id, { onDelete: "cascade" }),
  toCrewProfileId: uuid("to_crew_profile_id").notNull().references(() => crewProfiles.id, { onDelete: "cascade" }),

  rating: integer("rating").notNull(), // 1..5
  review: text("review"),

  createdAt: timestamp("created_at", { withTimezone: true }).notNull().defaultNow(),
});


Unieke constraint (1 sollicitatie per profiel per request) doe je het liefst in migration. Als je geen migration-generator gebruikt, kun je dit later via SQL toevoegen.

3) Storage layer (server/storage.ts)

Voeg methods toe (pseudo-implementatie met Drizzle). Pas import paden aan aan jouw project.

// server/storage.ts
import { db } from "./db";
import { and, eq, desc } from "drizzle-orm";
import { crewProfiles, crewRequests, crewApplications, businesses } from "../shared/schema";

export const storage = {
  async getCrewProfileByUserId(userId: string) {
    const [row] = await db.select().from(crewProfiles).where(eq(crewProfiles.userId, userId));
    return row ?? null;
  },

  async upsertCrewProfile(userId: string, data: any) {
    const existing = await this.getCrewProfileByUserId(userId);
    if (existing) {
      const [row] = await db
        .update(crewProfiles)
        .set({ ...data, updatedAt: new Date() })
        .where(eq(crewProfiles.id, existing.id))
        .returning();
      return row;
    }
    const [row] = await db
      .insert(crewProfiles)
      .values({ userId, ...data })
      .returning();
    return row;
  },

  async createCrewRequest(data: any) {
    const [row] = await db.insert(crewRequests).values(data).returning();
    return row;
  },

  async listCrewRequests(regionId?: string, category?: string) {
    // MVP: simpele filtering
    let q = db.select().from(crewRequests).where(eq(crewRequests.status, "open"));
    if (regionId) q = q.where(and(eq(crewRequests.status, "open"), eq(crewRequests.regionId, regionId)));
    // category filter in code (simpel) of met and() als je wilt
    const rows = await q.orderBy(desc(crewRequests.createdAt));
    return category ? rows.filter(r => r.category === category) : rows;
  },

  async listMyCrewRequests(businessId: string) {
    return db.select().from(crewRequests).where(eq(crewRequests.businessId, businessId)).orderBy(desc(crewRequests.createdAt));
  },

  async applyToCrewRequest(requestId: string, crewProfileId: string, message?: string) {
    // MVP: unieke constraint later; hier manual check
    const existing = await db
      .select()
      .from(crewApplications)
      .where(and(eq(crewApplications.requestId, requestId), eq(crewApplications.crewProfileId, crewProfileId)));
    if (existing.length) return existing[0];

    const [row] = await db
      .insert(crewApplications)
      .values({ requestId, crewProfileId, message: message ?? null })
      .returning();
    return row;
  },

  async listApplicationsForRequest(requestId: string) {
    return db.select().from(crewApplications).where(eq(crewApplications.requestId, requestId)).orderBy(desc(crewApplications.createdAt));
  },

  async updateApplicationStatus(applicationId: string, status: string) {
    const [row] = await db
      .update(crewApplications)
      .set({ status, updatedAt: new Date() })
      .where(eq(crewApplications.id, applicationId))
      .returning();
    return row ?? null;
  },

  async closeCrewRequest(requestId: string, status: "closed" | "filled" | "cancelled") {
    const [row] = await db
      .update(crewRequests)
      .set({ status, updatedAt: new Date() })
      .where(eq(crewRequests.id, requestId))
      .returning();
    return row ?? null;
  },
};

4) Routes (server/routes.ts)

Voeg deze endpoints toe. Ik ga uit van: req.user.id bestaat als je ingelogd bent, en jij hebt al getBusinessProfileByUserId of iets soortgelijks. Zo niet: we pakken businesses via owner id.

// server/routes.ts
import { z } from "zod";
import { storage } from "./storage";

const requireAuth = (req: any, res: any) => {
  if (!req.user?.id) {
    res.status(401).json({ error: "Niet ingelogd" });
    return false;
  }
  return true;
};

const crewProfileSchema = z.object({
  regionId: z.string().uuid(),
  displayName: z.string().min(2),
  headline: z.string().optional(),
  categories: z.array(z.string()).default([]),
  skills: z.array(z.string()).default([]),
  availability: z.record(z.any()).default({}),
  rateType: z.enum(["hour", "day", "fixed"]).default("hour"),
  rateMinEur: z.number().optional(),
  phone: z.string().optional(),
  bio: z.string().optional(),
  isActive: z.boolean().default(true),
});

const crewRequestSchema = z.object({
  regionId: z.string().uuid(),
  title: z.string().min(3),
  description: z.string().optional(),
  category: z.string().min(2),
  skills: z.array(z.string()).default([]),
  startAt: z.string(), // ISO
  endAt: z.string(),   // ISO
  rateType: z.enum(["hour", "day", "fixed", "negotiable"]).default("negotiable"),
  rateEur: z.number().optional(),
  locationText: z.string().optional(),
});

// Professional: get/me + upsert
app.get("/api/crew/me", async (req, res) => {
  if (!requireAuth(req, res)) return;
  const profile = await storage.getCrewProfileByUserId(req.user.id);
  res.json(profile);
});

app.post("/api/crew/me", async (req, res) => {
  if (!requireAuth(req, res)) return;
  const data = crewProfileSchema.parse(req.body);

  const saved = await storage.upsertCrewProfile(req.user.id, {
    ...data,
    rateMinEur: data.rateMinEur ?? null,
    phone: data.phone ?? null,
    bio: data.bio ?? null,
    headline: data.headline ?? null,
  });
  res.json(saved);
});

// List open requests (for everyone logged in)
app.get("/api/crew/requests", async (req, res) => {
  if (!requireAuth(req, res)) return;
  const { regionId, category } = req.query as any;
  const rows = await storage.listCrewRequests(regionId, category);
  res.json(rows);
});

// Business creates request
app.post("/api/crew/requests", async (req, res) => {
  if (!requireAuth(req, res)) return;

  // TODO: haal businessId van ingelogde user (via bestaande business profile route)
  // Voor MVP: verwacht businessId in body (maar liever niet)
  // Beter: storage.getBusinessByOwner(req.user.id)
  const business = await storage.getBedrijfsprofielByUserId?.(req.user.id);
  if (!business) return res.status(400).json({ error: "Maak eerst je bedrijfsprofiel aan." });

  const data = crewRequestSchema.parse(req.body);
  const created = await storage.createCrewRequest({
    businessId: business.id,
    regionId: data.regionId,
    title: data.title,
    description: data.description ?? null,
    category: data.category,
    skills: data.skills,
    startAt: new Date(data.startAt),
    endAt: new Date(data.endAt),
    rateType: data.rateType,
    rateEur: data.rateEur ?? null,
    locationText: data.locationText ?? null,
  });

  res.json(created);
});

// Business: my requests
app.get("/api/crew/requests/mine", async (req, res) => {
  if (!requireAuth(req, res)) return;
  const business = await storage.getBedrijfsprofielByUserId?.(req.user.id);
  if (!business) return res.json([]);
  const rows = await storage.listMyCrewRequests(business.id);
  res.json(rows);
});

// Apply
app.post("/api/crew/requests/:id/apply", async (req, res) => {
  if (!requireAuth(req, res)) return;

  const profile = await storage.getCrewProfileByUserId(req.user.id);
  if (!profile) return res.status(400).json({ error: "Maak eerst je RegioCrew-profiel aan." });

  const message = typeof req.body?.message === "string" ? req.body.message : "";
  const appRow = await storage.applyToCrewRequest(req.params.id, profile.id, message);
  res.json(appRow);
});

// Business: applications for a request
app.get("/api/crew/requests/:id/applications", async (req, res) => {
  if (!requireAuth(req, res)) return;

  // MVP: geen ownership check hier? liever wel:
  // 1) load request, 2) check businessId == my business.id
  const business = await storage.getBedrijfsprofielByUserId?.(req.user.id);
  if (!business) return res.status(403).json({ error: "Geen business" });

  // TODO: ownership check toevoegen (bulletproof)
  const rows = await storage.listApplicationsForRequest(req.params.id);
  res.json(rows);
});

// Update application status
app.post("/api/crew/applications/:id/status", async (req, res) => {
  if (!requireAuth(req, res)) return;

  const status = z.enum(["shortlisted", "accepted", "rejected"]).parse(req.body?.status);
  const updated = await storage.updateApplicationStatus(req.params.id, status);
  res.json(updated);
});

// Close request
app.post("/api/crew/requests/:id/close", async (req, res) => {
  if (!requireAuth(req, res)) return;
  const status = z.enum(["closed", "filled", "cancelled"]).parse(req.body?.status);
  const updated = await storage.closeCrewRequest(req.params.id, status);
  res.json(updated);
});

5) UI pagina’s (minimaal, werkt meteen)

Je maakt 4 routes:

/regiocrew → lijst open requests + 2 knoppen

/regiocrew/new → request plaatsen

/regiocrew/profile → professional profiel

/regiocrew/requests/:id → detail + apply / applications

Data hooks: je gebruikt al react-query met apiRequest.

Voorbeeld: lijst requests (client/src/pages/regiocrew.tsx):

import { useQuery } from "@tanstack/react-query";
import { Link } from "wouter";

export default function RegioCrew() {

      </div>

      <div className="grid md:grid-cols-2 gap-4">

            <div className="font-medium">{r.title}</div>
            <div className="text-sm text-muted-foreground">{r.category} • {r.locationText ?? "—"}</div>
            <div className="text-sm mt-2">
              {new Date(r.startAt).toLocaleString()} → {new Date(r.endAt).toLocaleString()}
            </div>
          </Link>
        ))}
      </div>
    </div>
  );
}
