2) DB: maak categorieën tabel + koppel aan requests + documents

Run dit in je Postgres (Replit DB/Neon/Supabase).

2.1 Maak tabel woo_categories
create table if not exists woo_categories (
  slug text primary key,
  label text not null,
  is_allowed boolean not null default true
);

-- seed (idempotent)
insert into woo_categories (slug, label, is_allowed) values
  ('mandaat_delegatie', 'Mandaat & delegatie', true),
  ('beleid_verordening', 'Beleid & verordeningen', true),
  ('vergunningen', 'Vergunningen & beleidsregels', true),
  ('heffingen_leges', 'Heffingen, leges, belastingen', true),
  ('handhaving_kaders', 'Handhavingskaders (beleid)', true),
  ('aanbesteding', 'Aanbestedingen & gunning', true),
  ('subsidies', 'Subsidiekaders & besluiten', true),
  ('uitvoering_partijen', 'Uitvoeringsorganisaties/derden', true),
  ('openbaarheid_archief', 'Archief/openbaarheid/werkinstructies', true),
  ('persoonlijk_verkeer_boete', 'Persoonlijk/Verkeer/Boete (NIET TOEGESTAAN)', false)
on conflict (slug) do update
set label = excluded.label,
    is_allowed = excluded.is_allowed;

2.2 Voeg category_slug toe aan woo_requests (verplicht)
alter table woo_requests
add column if not exists category_slug text;

-- tijdelijke default (kies wat past)
update woo_requests
set category_slug = 'beleid_verordening'
where category_slug is null;

alter table woo_requests
alter column category_slug set not null;

alter table woo_requests
add constraint fk_woo_requests_category
foreign key (category_slug) references woo_categories(slug);

2.3 Voeg optional category toe aan woo_documents

(soms wil je bijlagen anders labelen)

alter table woo_documents
add column if not exists category_slug text;

-- default naar request category als leeg (alleen als je wilt)
update woo_documents d
set category_slug = r.category_slug
from woo_requests r
where d.request_id = r.id and d.category_slug is null;

alter table woo_documents
add constraint fk_woo_documents_category
foreign key (category_slug) references woo_categories(slug);

2.4 HARD BLOCK: alleen allowed categories mogen in requests

Dit is de echte guard.

create or replace function enforce_allowed_woo_category()
returns trigger as $$
begin
  if not exists (
    select 1 from woo_categories c
    where c.slug = new.category_slug and c.is_allowed = true
  ) then
    raise exception 'Category % is not allowed in OpenRegio WOO library', new.category_slug;
  end if;
  return new;
end;
$$ language plpgsql;

drop trigger if exists trg_enforce_allowed_woo_category on woo_requests;

create trigger trg_enforce_allowed_woo_category
before insert or update on woo_requests
for each row execute function enforce_allowed_woo_category();

3) API: server-side validatie (extra veilig)

DB is leidend, maar API moet netjes foutmelding geven.

3.1 Endpoint: “haal categorieën op”

Voeg toe:

GET /api/woo/categories → alleen allowed categories tonen

SQL:

select slug, label from woo_categories where is_allowed = true order by label;
