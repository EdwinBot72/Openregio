# OpenRegio — Pijler 2 Lokale Vindbaarheid (Replit Build Instructies)

## Wat te bouwen

Vervang de huidige Pijler 2 "zichtbaarheid op het platform" sectie door een nieuwe module:
**"Gevonden worden in jouw regio"** — een interactieve tool die lokale ondernemers leert
welke zoektermen, websiteteksten en SEO-elementen zij nodig hebben om via Google en
AI-zoekmachines gevonden te worden in hun regio.

---

## Bestandsstructuur — maak deze bestanden aan

```
client/src/
  components/
    lokale-vindbaarheid/
      LokaleVindbaarheid.tsx        ← hoofd-component (tabs wrapper)
      ZoektermenFinder.tsx          ← Tab 1: zoektermen generator
      WebsiteChecklist.tsx          ← Tab 2: SEO checklist met voortgang
      GoogleEnAI.tsx                ← Tab 3: Google Bedrijfsprofiel + AI-zoekers
      VoorbeeldenGalerij.tsx        ← Tab 4: voorbeelden per beroep
      types.ts                      ← gedeelde TypeScript types

  pages/
    dashboard/pijler-twee.tsx       ← pagina die LokaleVindbaarheid laadt

server/
  routes/
    zoektermen.ts                   ← POST /api/zoektermen/genereer
    checklist.ts                    ← GET/POST /api/checklist/:userId

shared/
  seo-data.ts                       ← beroep-data, voorbeeldtermen, tips
```

## 3. `server/routes/zoektermen.ts`

```typescript
import { Router } from "express";
import { BEROEP_DATA } from "../../shared/seo-data";
import type { ZoektermenRequest, ZoektermenResponse } from "../../shared/types";

const router = Router();

// POST /api/zoektermen/genereer
router.post("/genereer", (req, res) => {
  const { beroep, stad, wijk } = req.body as ZoektermenRequest;

  if (!beroep || !stad) {
    return res.status(400).json({ error: "beroep en stad zijn verplicht" });
  }

  const data = BEROEP_DATA[beroep as keyof typeof BEROEP_DATA];
  if (!data) {
    return res.status(400).json({ error: "onbekend beroep" });
  }

  const fill = (s: string) =>
    s
      .replace(/{beroep}/g, beroep)
      .replace(/{stad}/g, stad)
      .replace(/{wijk}/g, wijk ?? stad);

  const response: ZoektermenResponse = {
    primair: data.primaireTermen.map(fill),
    longTail: data.longTail.map(fill),
    zoekvragen: data.zoekvragen.map(fill),
    paginatitel: `${data.label} ${stad}${wijk ? " " + wijk : ""} | [Jouw bedrijfsnaam]`,
    metaDescription: `Uw ${beroep} in ${stad}${wijk ? " " + wijk : ""}. ${data.primaireTermen[1] ? fill(data.primaireTermen[1]) : ""} — Bel direct voor een afspraak.`,
    h1Suggestie: `${data.label} in ${stad} — [Jouw slogan of specialiteit]`,
    wijkTip: data.wijkTip,
  };

  res.json(response);
});

export default router;
```

## 4. `server/routes/checklist.ts`

```typescript
import { Router } from "express";
import { db } from "../db"; // gebruik bestaande db connectie

const router = Router();

// GET /api/checklist/:userId
router.get("/:userId", async (req, res) => {
  try {
    const result = await db.query(
      "SELECT afgevinkt FROM seo_checklist WHERE user_id = $1",
      [req.params.userId]
    );
    res.json({ afgevinkt: result.rows[0]?.afgevinkt ?? [] });
  } catch {
    res.json({ afgevinkt: [] });
  }
});

// POST /api/checklist/:userId
router.post("/:userId", async (req, res) => {
  const { afgevinkt } = req.body as { afgevinkt: string[] };
  try {
    await db.query(
      `INSERT INTO seo_checklist (user_id, afgevinkt, updated_at)
       VALUES ($1, $2, NOW())
       ON CONFLICT (user_id) DO UPDATE
       SET afgevinkt = $2, updated_at = NOW()`,
      [req.params.userId, JSON.stringify(afgevinkt)]
    );
    res.json({ ok: true });
  } catch (err) {
    res.status(500).json({ error: "opslaan mislukt" });
  }
});

export default router;
```

## 8. `client/src/components/lokale-vindbaarheid/ZoektermenFinder.tsx`

```tsx
import { useState } from "react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Badge } from "@/components/ui/badge";
import { Card, CardContent } from "@/components/ui/card";
import { BEROEP_DATA, type Beroep } from "@/shared/seo-data";
import type { ZoektermenResponse } from "@/shared/types";

export default function ZoektermenFinder() {
  const [gekozenBeroep, setGekozenBeroep] = useState<Beroep | "">("");
  const [stad, setStad] = useState("");
  const [resultaat, setResultaat] = useState<ZoektermenResponse | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  const beroepen = Object.entries(BEROEP_DATA) as [Beroep, (typeof BEROEP_DATA)[Beroep]][];

  async function genereer() {
    if (!gekozenBeroep) { setError("Kies eerst een beroep."); return; }
    if (!stad.trim()) { setError("Vul een stad of regio in."); return; }
    setError("");
    setLoading(true);
    try {
      const res = await fetch("/api/zoektermen/genereer", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ beroep: gekozenBeroep, stad: stad.trim() }),
      });
      const data = await res.json();
      setResultaat(data);
    } catch {
      setError("Kon zoektermen niet ophalen. Probeer opnieuw.");
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="space-y-5 pt-4">
      {/* Beroep kiezen */}
      <div>
        <p className="text-sm font-medium text-gray-700 mb-2">
          Stap 1 — Wat voor ondernemer ben je?
        </p>
        <div className="flex flex-wrap gap-2">
          {beroepen.map(([key, val]) => (
            <button
              key={key}
              onClick={() => setGekozenBeroep(key)}
              className={`px-3 py-1.5 rounded-lg border text-sm transition-all ${
                gekozenBeroep === key
                  ? "bg-emerald-700 border-emerald-700 text-white font-medium"
                  : "border-gray-200 text-gray-600 hover:border-emerald-400 hover:text-emerald-700"
              }`}
            >
              {val.emoji} {val.label}
            </button>
          ))}
        </div>
      </div>

      {/* Stad invullen */}
      <div>
        <p className="text-sm font-medium text-gray-700 mb-2">
          Stap 2 — Jouw stad of regio
        </p>
        <div className="flex gap-2 max-w-md">
          <Input
            placeholder="bijv. Haarlem, Tilburg, Groningen..."
            value={stad}
            onChange={(e) => setStad(e.target.value)}
            onKeyDown={(e) => e.key === "Enter" && genereer()}
          />
          <Button onClick={genereer} disabled={loading} className="bg-emerald-700 hover:bg-emerald-800">
            {loading ? "..." : "Genereer"}
          </Button>
        </div>
        {error && <p className="text-sm text-red-500 mt-1">{error}</p>}
      </div>

      {/* Resultaten */}
      {resultaat && (
        <div className="space-y-4">
          <Card>
            <CardContent className="pt-4 space-y-4">
              {/* Paginatitel suggestie */}
              <div>
                <p className="text-xs font-medium text-gray-400 uppercase tracking-wide mb-1">
                  Aanbevolen paginatitel
                </p>
                <p className="text-sm font-mono bg-gray-50 rounded px-3 py-2 border">
                  {resultaat.paginatitel}
                </p>
              </div>

              {/* Meta description */}
              <div>
                <p className="text-xs font-medium text-gray-400 uppercase tracking-wide mb-1">
                  Meta description
                </p>
                <p className="text-sm font-mono bg-gray-50 rounded px-3 py-2 border">
                  {resultaat.metaDescription}
                </p>
              </div>

              {/* H1 */}
              <div>
                <p className="text-xs font-medium text-gray-400 uppercase tracking-wide mb-1">
                  H1 suggestie
                </p>
                <p className="text-sm font-mono bg-gray-50 rounded px-3 py-2 border">
                  {resultaat.h1Suggestie}
                </p>
              </div>
            </CardContent>
          </Card>

          <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
            <TermenKaart
              titel="Primaire zoektermen"
              termen={resultaat.primair}
              kleur="emerald"
            />
            <TermenKaart
              titel="Long-tail kansen"
              termen={resultaat.longTail}
              kleur="blue"
            />
            <TermenKaart
              titel="Zoekvragen"
              termen={resultaat.zoekvragen}
              kleur="amber"
            />
          </div>

          {resultaat.wijkTip && (
            <div className="bg-emerald-50 border border-emerald-100 rounded-lg p-4">
              <p className="text-sm font-medium text-emerald-800 mb-1">
                💡 Tip voor jouw beroep

}

function TermenKaart({
  titel,
  termen,
  kleur,
}: {
  titel: string;
  termen: string[];
  kleur: "emerald" | "blue" | "amber";
```

## 9. `client/src/components/lokale-vindbaarheid/WebsiteChecklist.tsx`

```tsx
import { useState, useEffect } from "react";
import { CHECKLIST_ITEMS } from "@/shared/seo-data";
import { useAuth } from "@/hooks/use-auth"; // bestaande auth hook
import { Badge } from "@/components/ui/badge";

const PRIORITEIT_KLEUR = {
  hoog: "bg-red-50 text-red-600 border-red-100",
  midden: "bg-amber-50 text-amber-600 border-amber-100",
  laag: "bg-gray-50 text-gray-500 border-gray-100",
};

export default function WebsiteChecklist() {
  const { user } = useAuth();
  const [afgevinkt, setAfgevinkt] = useState<Set<string>>(new Set());

  useEffect(() => {
    if (!user) return;
    fetch(`/api/checklist/${user.id}`)
      .then((r) => r.json())
      .then((d) => setAfgevinkt(new Set(d.afgevinkt ?? [])));
  }, [user?.id]);

  async function toggle(id: string) {
    const nieuw = new Set(afgevinkt);
    nieuw.has(id) ? nieuw.delete(id) : nieuw.add(id);
    setAfgevinkt(nieuw);
    if (user) {
      await fetch(`/api/checklist/${user.id}`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ afgevinkt: [...nieuw] }),
      });
    }
  }

  const voortgang = Math.round((afgevinkt.size / CHECKLIST_ITEMS.length) * 100);

  return (
    <div className="space-y-4 pt-4">
      {/* Voortgangsbalk */}
      <div>
        <div className="flex justify-between text-sm mb-1">
          <span className="text-gray-500">Voortgang</span>
          <span className="font-medium text-emerald-700">
            {afgevinkt.size}/{CHECKLIST_ITEMS.length} punten
          </span>
        </div>
        <div className="h-2 bg-gray-100 rounded-full overflow-hidden">
          <div
            className="h-full bg-emerald-600 rounded-full transition-all duration-300"
            style={{ width: `${voortgang}%` }}
          />
        </div>
      </div>

      {/* Checklist */}
      <div className="space-y-2">
        {CHECKLIST_ITEMS.map((item) => {
          const gedaan = afgevinkt.has(item.id);
          return (
            <button
              key={item.id}
              onClick={() => toggle(item.id)}
              className={`w-full text-left flex items-start gap-3 p-3 rounded-lg border transition-all ${
                gedaan
                  ? "bg-emerald-50 border-emerald-100"
                  : "bg-white border-gray-100 hover:border-emerald-200"
              }`}
            >
              <div
                className={`w-5 h-5 rounded-full border-2 flex-shrink-0 mt-0.5 flex items-center justify-center transition-all ${
                  gedaan
                    ? "bg-emerald-600 border-emerald-600"
                    : "border-gray-300"
                }`}
              >
                {gedaan && (
                  <svg className="w-3 h-3 text-white" fill="none" viewBox="0 0 12 12">
                    <path
                      d="M2 6l3 3 5-5"
                      stroke="currentColor"
                      strokeWidth="2"
                      strokeLinecap="round"
                      strokeLinejoin="round"
                    />
                  </svg>
                )}
              </div>
              <div className="flex-1 min-w-0">
                <div className="flex items-center gap-2 flex-wrap">
                  <span
                    className={`text-sm font-medium ${
                      gedaan ? "line-through text-gray-400" : "text-gray-800"
                    }`}
                  >
                    {item.titel}
                  </span>
                  <span
                    className={`text-xs px-1.5 py-0.5 rounded border ${
                      PRIORITEIT_KLEUR[item.prioriteit]
                    }`}
                  >
                    {item.prioriteit}
                  </span>
                </div>
                <p
                  className={`text-xs mt-0.5 ${
                    gedaan ? "text-gray-300" : "text-gray-400"
                  }`}
                >
                  {item.uitleg}
                </p>
              </div>
            </button>
          );
        })}
      </div>
    </div>
  );
}
```
